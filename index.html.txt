    // Codificación UTF-8 segura para Base64 (soporta tildes y ñ)
    function utf8ToB64(str) {
      const bytes = new TextEncoder().encode(str);
      let binString = '';
      bytes.forEach((b) => { binString += String.fromCharCode(b); });
      return btoa(binString);
    }

    async function guardarEnGitHub(silencioso = false) {
      localStorage.setItem('padelrank_cache', JSON.stringify(db));
      renderAll();

      if (!ghConfig.token || !ghConfig.owner || !ghConfig.repo) {
        if (!silencioso) showToast('Guardado en local. Configura tu Token de GitHub para subir a la nube.', 'warn');
        return;
      }

      const apiUrl = `https://api.github.com/repos/${ghConfig.owner}/${ghConfig.repo}/contents/${ghConfig.path}`;
      const headers = {
        'Authorization': `Bearer ${ghConfig.token}`,
        'Accept': 'application/vnd.github+json',
        'Content-Type': 'application/json'
      };

      try {
        // 1. Obtener el SHA actual del archivo datos.json (si ya existe)
        let sha = null;
        const getRes = await fetch(`${apiUrl}?ref=${ghConfig.branch}&t=${Date.now()}`, { headers });
        if (getRes.ok) {
          const fileInfo = await getRes.json();
          sha = fileInfo.sha;
        }

        // 2. Hacer PUT con el nuevo JSON formateado
        const contentB64 = utf8ToB64(JSON.stringify(db, null, 2));
        const body = {
          message: `Actualización PadelRank (${new Date().toLocaleString('es-CL')})`,
          content: contentB64,
          branch: ghConfig.branch
        };
        if (sha) body.sha = sha;

        const putRes = await fetch(apiUrl, {
          method: 'PUT',
          headers,
          body: JSON.stringify(body)
        });

        if (!putRes.ok) {
          const errData = await putRes.json();
          throw new Error(errData.message || 'Error al escribir en GitHub');
        }

        document.getElementById('syncBadge').innerHTML = `<span class="w-2 h-2 rounded-full bg-neon"></span> Guardado en GitHub ✓`;
        showToast('✅ Base de datos actualizada en GitHub', 'ok');
      } catch (err) {
        console.error(err);
        showToast(`❌ Error GitHub: ${err.message}`, 'err');
      }
    }

    function getEquipoPuntos(equipo) {
      return equipo.jugadorIds.reduce((acc, jId) => {
        const j = db.jugadores.find(p => p.id === jId && p.activo);
        return acc + (j ? Number(j.puntos) : 0);
      }, 0);
    }

    function calcularVariacionPartido(eqGanador, eqPerdedor) {
      const ptsWin = getEquipoPuntos(eqGanador);
      const ptsLose = getEquipoPuntos(eqPerdedor);
      const cfg = db.config;

      let bonoBatacazo = 0;
      if (ptsWin < ptsLose) {
        bonoBatacazo = (ptsLose - ptsWin) * Number(cfg.multiplicadorBatacazo);
      }

      const deltaGanador = Number(cfg.puntosVictoriaBase) + bonoBatacazo;
      const deltaPerdedor = -(Number(cfg.puntosDerrotaBase) + bonoBatacazo);

      return { ptsWin, ptsLose, bonoBatacazo, deltaGanador, deltaPerdedor };
    }

    function previsualizarPuntos() {
      const wId = document.getElementById('matchWinner').value;
      const lId = document.getElementById('matchLoser').value;
      const box = document.getElementById('previewCalculo');
      if (!wId || !lId || wId === lId) {
        box.innerHTML = 'Selecciona dos equipos distintos para previsualizar el puntaje.';
        return;
      }
      const eqW = db.equipos.find(e => e.id === wId);
      const eqL = db.equipos.find(e => e.id === lId);
      const calc = calcularVariacionPartido(eqW, eqL);

      box.innerHTML = `
        <div class="flex flex-wrap justify-between gap-2">

        </div>
        ${calc.bonoBatacazo > 0 ? `<p class="text-amber-300 mt-1">🔥 ¡Batacazo detectado! Bono extra de ${calc.bonoBatacazo} pts por vencer a un equipo con mayor puntaje.</p>` : ''}
      `;
    }

    async function registrarPartido(e) {
      e.preventDefault();
      const wId = document.getElementById('matchWinner').value;
      const lId = document.getElementById('matchLoser').value;
      const score = document.getElementById('matchScore').value.trim();
      const fecha = document.getElementById('matchDate').value;

      if (wId === lId) {
        showToast('El equipo ganador y perdedor no pueden ser el mismo', 'err');
        return;
      }

      const eqW = db.equipos.find(eq => eq.id === wId);
      const eqL = db.equipos.find(eq => eq.id === lId);
      const calc = calcularVariacionPartido(eqW, eqL);

      // Aplicar puntos a cada integrante activo del equipo ganador
      eqW.jugadorIds.forEach(jId => {
        const j = db.jugadores.find(p => p.id === jId && p.activo);
        if (j) {
          j.puntos += calc.deltaGanador;
          j.pj += 1;
          j.pg += 1;
        }
      });

      // Aplicar puntos a cada integrante activo del equipo perdedor
      eqL.jugadorIds.forEach(jId => {
        const j = db.jugadores.find(p => p.id === jId && p.activo);
        if (j) {
          j.puntos += calc.deltaPerdedor;
          if (!db.config.permitirNegativos && j.puntos < 0) j.puntos = 0;
          j.pj += 1;
          j.pp += 1;
        }
      });

      db.partidos.unshift({
        id: 'm' + Date.now(),
        fecha,
        ganadorId: eqW.id,
        ganadorNombre: eqW.nombre,
        perdedorId: eqL.id,
        perdedorNombre: eqL.nombre,
        marcador: score,
        deltaGanador: calc.deltaGanador,
        deltaPerdedor: calc.deltaPerdedor,
        ganadorJugadorIds: [...eqW.jugadorIds],
        perdedorJugadorIds: [...eqL.jugadorIds]
      });

      document.getElementById('matchScore').value = '';
      await guardarEnGitHub();
    }

    async function anularPartido(matchId) {
      if (!confirm('¿Anular este partido y revertir los puntos sumados/restados a los jugadores?')) return;
      const idx = db.partidos.findIndex(m => m.id === matchId);
      if (idx === -1) return;
      const m = db.partidos[idx];

      (m.ganadorJugadorIds || []).forEach(jId => {
        const j = db.jugadores.find(p => p.id === jId);
        if (j) {
          j.puntos -= m.deltaGanador;
          j.pj = Math.max(0, j.pj - 1);
          j.pg = Math.max(0, j.pg - 1);
        }
      });

      (m.perdedorJugadorIds || []).forEach(jId => {
        const j = db.jugadores.find(p => p.id === jId);
        if (j) {
          j.puntos -= m.deltaPerdedor;
          j.pj = Math.max(0, j.pj - 1);
          j.pp = Math.max(0, j.pp - 1);
        }
      });

      db.partidos.splice(idx, 1);
      await guardarEnGitHub();
    }

            <td class="py-3 px-4 font-semibold text-white">${j.nombre}</td>
            <td class="py-3 px-4 text-xs text-slate-300">${eq ? eq.nombre : 'Sin equipo'}</td>
            <td class="py-3 px-4 text-center text-slate-300">${j.pj}</td>
            <td class="py-3 px-4 text-center text-xs"><span class="text-neon">${j.pg}G</span> / <span class="text-rose-400">${j.pp}P</span></td>
            <td class="py-3 px-4 text-right font-mono font-bold text-white">${j.puntos}</td>
          </tr>
        `;
      }).join('');

      // Historial de Partidos
      const listaPartidos = document.getElementById('listaPartidos');
      if (db.partidos.length === 0) {
        listaPartidos.innerHTML = `<p class="text-sm text-slate-400 py-6 text-center">Aún no hay partidos registrados. Todos comienzan en 0 puntos.</p>`;
      } else {
        listaPartidos.innerHTML = db.partidos.map(m => `
          <div class="bg-slate-900/80 border border-slate-700/70 rounded-xl p-4 flex flex-wrap items-center justify-between gap-3">
            <div>
              <div class="text-xs text-slate-400 mb-1">📅 ${m.fecha} • Marcador: <strong class="text-white">${m.marcador}</strong></div>
              <div class="text-sm">
                <span class="font-bold text-neon">🏆 ${m.ganadorNombre} (+${m.deltaGanador} pts)</span>
                <span class="text-slate-500 mx-2">vs</span>
                <span class="font-semibold text-rose-400">${m.perdedorNombre} (${m.deltaPerdedor} pts)</span>
              </div>
            </div>
            ${isAdmin ? `<button onclick="anularPartido('${m.id}')" class="px-3 py-1.5 rounded-lg bg-rose-950 hover:bg-rose-900 text-rose-200 border border-rose-800 text-xs font-semibold">↩️ Anular Partido</button>` : ''}
          </div>
        `).join('');
      }

      // Selectores Admin
      const optsEquipos = db.equipos.filter(e => e.activo).map(e => `<option value="${e.id}">${e.nombre} (${getEquipoPuntos(e)} pts)</option>`).join('');
      document.getElementById('matchWinner').innerHTML = optsEquipos;
      document.getElementById('matchLoser').innerHTML = optsEquipos;
      if (db.equipos.length >= 2) {
        document.getElementById('matchLoser').selectedIndex = 1;
      }
      previsualizarPuntos();

      const optsJugadores = db.jugadores.filter(j => j.activo).map(j => `<option value="${j.id}">${j.nombre} (${j.puntos} pts)</option>`).join('');
      document.getElementById('newTeamP1').innerHTML = optsJugadores;
      document.getElementById('newTeamP2').innerHTML = optsJugadores;

      // Admin Lista Equipos
      document.getElementById('adminListaEquipos').innerHTML = db.equipos.filter(e => e.activo).map(eq => {
        const integrantesHtml = eq.jugadorIds.map(id => {
          const j = db.jugadores.find(p => p.id === id);
          if (!j) return '';
          return `
            <div class="flex justify-between items-center bg-slate-900 px-3 py-1.5 rounded-lg text-xs">
              <span>${j.nombre} (${j.puntos} pts)</span>
              <button onclick="quitarIntegranteEquipo('${eq.id}', '${j.id}')" class="text-rose-400 hover:underline">Quitar</button>
            </div>
          `;
        }).join('');

        const disponibles = db.jugadores.filter(j => j.activo && !eq.jugadorIds.includes(j.id));
        const addHtml = disponibles.length > 0 ? `
          <div class="flex gap-2 mt-2">
            <select id="addMember-${eq.id}" class="flex-1 px-2 py-1.5 rounded-lg bg-slate-900 border border-slate-700 text-xs text-white">
              ${disponibles.map(d => `<option value="${d.id}">${d.nombre}</option>`).join('')}
            </select>
            <button onclick="agregarIntegranteEquipo('${eq.id}')" class="px-3 py-1.5 rounded-lg bg-slate-800 hover:bg-slate-700 text-neon text-xs font-bold border border-slate-600">+ Sumar</button>
          </div>
        ` : `<p class="text-[11px] text-slate-500 mt-2">No hay más jugadores disponibles para agregar.</p>`;

        return `
          <div class="bg-slate-950/60 border border-slate-700/80 rounded-xl p-4 space-y-2">
            <div class="flex justify-between items-center">
              <h4 class="font-bold text-white">${eq.nombre}</h4>
              <span class="text-xs font-mono text-neon font-bold">${getEquipoPuntos(eq)} pts</span>
            </div>
            <div class="space-y-1.5">${integrantesHtml}</div>
            ${addHtml}
          </div>
        `;
      }).join('');

      // Admin Lista Jugadores
      document.getElementById('adminListaJugadores').innerHTML = db.jugadores.map(j => `
        <div class="py-3 flex flex-wrap items-center justify-between gap-2">
          <div>
            <span class="font-semibold text-white text-sm">${j.nombre}</span>
            <span class="ml-2 font-mono text-xs text-neon font-bold">${j.puntos} pts</span>
          </div>
          <div class="flex items-center gap-1.5">
            <button onclick="ajustarPuntosJugador('${j.id}', -50)" class="px-2.5 py-1 rounded-lg bg-slate-800 hover:bg-rose-950 text-rose-300 text-xs font-bold">-50</button>
            <button onclick="ajustarPuntosJugador('${j.id}', 50)" class="px-2.5 py-1 rounded-lg bg-slate-800 hover:bg-neon/20 text-neon text-xs font-bold">+50</button>
          </div>
        </div>
      `).join('');

      // Admin Lista Usuarios
      document.getElementById('adminListaUsuarios').innerHTML = db.usuarios.map(u => `
        <div class="py-3 flex flex-wrap items-center justify-between gap-2">
          <div>
            <div class="font-semibold text-white text-sm">${u.nombre} <span class="text-xs text-slate-400">(${u.email})</span></div>
            <div class="text-xs mt-0.5">
              Rol: <strong class="${u.rol === 'admin' ? 'text-neon' : 'text-sky-400'} uppercase">${u.rol}</strong> •
              Estado: <span class="${u.activo ? 'text-emerald-400' : 'text-rose-400'}">${u.activo ? 'Activo' : 'Inactivo'}</span>
            </div>
          </div>
          <div class="flex items-center gap-2">
            <button onclick="cambiarRolUsuario('${u.id}')" class="px-3 py-1 rounded-lg bg-slate-800 hover:bg-slate-700 text-xs text-white border border-slate-600">Cambiar Rol</button>
            <button onclick="toggleUsuarioActivo('${u.id}')" class="px-3 py-1 rounded-lg bg-slate-800 hover:bg-slate-700 text-xs text-slate-300 border border-slate-600">${u.activo ? 'Desactivar' : 'Activar'}</button>
          </div>
        </div>
      `).join('');

      // Cargar Reglas en Formulario
      document.getElementById('cfgNombre').value = db.config.torneoNombre;
      document.getElementById('cfgWinBase').value = db.config.puntosVictoriaBase;
      document.getElementById('cfgLossBase').value = db.config.puntosDerrotaBase;
      document.getElementById('cfgMult').value = db.config.multiplicadorBatacazo;
      document.getElementById('cfgNegativos').checked = db.config.permitirNegativos;
    }

    function showToast(msg, type = 'ok') {
      const el = document.getElementById('toast');
      el.textContent = msg;
      el.className = 'fixed bottom-5 right-5 z-50 max-w-sm px-4 py-3 rounded-xl shadow-2xl border text-sm font-medium transition-all';
      if (type === 'ok') el.classList.add('bg-slate-900', 'border-neon', 'text-neon');
      else if (type === 'warn') el.classList.add('bg-slate-900', 'border-amber-400', 'text-amber-300');
      else el.classList.add('bg-rose-950', 'border-rose-500', 'text-rose-200');
      el.classList.remove('hidden');
      setTimeout(() => el.classList.add('hidden'), 4000);
    }
